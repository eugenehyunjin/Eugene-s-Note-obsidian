CREATE TABLE IF NOT EXISTS `vehicle_tracking_api` (
    `idx`            BIGINT NOT NULL AUTO_INCREMENT,
    `comp_name`      VARCHAR(255) COLLATE utf8mb4_general_ci NOT NULL,
    `api_url`        VARCHAR(2048) COLLATE utf8mb4_general_ci NOT NULL,
    `api_key`        VARCHAR(255) COLLATE utf8mb4_general_ci DEFAULT NULL,
    `api_type`       VARCHAR(50) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT 'API 역할 (예: CAR_REGISTER, CAR_DELETE, CAR_STATUS)',
    `lot_area_no`    VARCHAR(50) COLLATE utf8mb4_general_ci DEFAULT NULL,
    `danji_code`     BIGINT DEFAULT NULL,
    `apt_code`       BIGINT DEFAULT NULL,
    `port_no`        INT DEFAULT NULL,
    `dc_count`       INT DEFAULT NULL,
    `created_at`     TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
    `updated_at`     TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
    PRIMARY KEY (`idx`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=UTF8MB4_GENERAL_CI;
