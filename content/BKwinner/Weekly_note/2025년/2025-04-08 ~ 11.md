  AND vh.in_date BETWEEN vv.scheduled_date AND vv.out_date
  AND vh.use_flag = "Y"
  AND vh.`type` = "visit"
  AND vh.out_date IN (
      SELECT out_date
      FROM vehicle_histories
      WHERE apartment_idx = '6084'
        AND out_date IS NOT NULL
        AND in_date > '2025-03-11'
        AND use_flag = "Y"
        AND `type` = "visit"
      GROUP BY out_date
      HAVING COUNT(*) > 1
  )
ORDER BY vh.out_date ASC;
