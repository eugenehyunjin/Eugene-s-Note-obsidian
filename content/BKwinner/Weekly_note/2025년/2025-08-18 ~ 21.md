import 'package:dio/dio.dart';  
import '../../../features/auth/apartment/data/apartment_api.dart';  
import '../../constants/api_constants.dart';  
import '../../storage/token_storage.dart';  
import '../../storage/user_storage.dart';  
import '../../storage/apartment_storage.dart';  
  
class AuthInterceptor extends Interceptor {  
  final Dio _authDio;  
  final ApartmentApi _apartmentApi;  
  
  AuthInterceptor(this._authDio, {ApartmentApi? apartmentApi})  
      : _apartmentApi = apartmentApi ?? ApartmentApi();  
  
  @override  
  void onRequest(RequestOptions options, RequestInterceptorHandler handler) async {  
    final authHeader = await TokenStorage.getAuthorizationHeader();  
    if (authHeader != null) {  
      options.headers['Authorization'] = authHeader;  
    }  
    handler.next(options);  
  }  
  
  @override  
  void onResponse(Response response, ResponseInterceptorHandler handler) async {  
    final path = response.requestOptions.path;  
    final ok = response.statusCode == 200;  
  
    if (ok && (path.contains(ApiConstants.signin) || path.contains(ApiConstants.kakaoLogin))) {  
      await _persistLoginPayload(response);  
    }  
  
    handler.next(response);  
  }  
  
  @override  
  void onError(DioException err, ErrorInterceptorHandler handler) {  
    handler.next(err);  
  }  
  
  Future<void> _persistLoginPayload(Response response) async {  
    try {  
      final result = response.data['result'] as Map<String, dynamic>?;  
  
      if (result == null) {  
        print('❌ 로그인 결과(result) 없음');  
        return;  
      }  
  
      // 1) 토큰 저장  
      final accessToken  = result['accessToken'] as String?;  
      final refreshToken = result['refreshToken'] as String? ?? '';  
      final tokenType    = result['tokenType'] as String? ?? 'Bearer';  
  
      if (accessToken == null) {  
        print('❌ accessToken 없음');  
        return;  
      }  
      await TokenStorage.saveTokens(accessToken, refreshToken, tokenType);  
      print('✅ 토큰 저장 완료');  
  
      // 2) 닉네임만 세션에 저장  
      final nickName = result['nickName'] as String?;  
      if (nickName != null) {  
        await UserStorage.saveNickName(nickName);  
        print('✅ 유저 닉네임 저장(세션) 완료');  
      }  
  
      // 3) 아파트 정보 처리 (BaseApi 사용)  
      final hasApt = (result['hasApartment'] as bool?) ?? false;  
      if (hasApt) {  
        await _fetchAndPersistApartmentInfo();  
      } else {  
        await ApartmentStorage.clear();  
        print('ℹ️ 아파트 미연결: 스토리지 초기화');  
      }  
    } catch (e) {  
      print('❌ 로그인 페이로드 저장 실패: $e');  
    }  
  }  
  
  Future<void> _fetchAndPersistApartmentInfo() async {  
    try {  
      // Authorization 헤더는 Interceptor로 자동 주입됨  
      final apt = await _apartmentApi.getUserApartment(); // BaseApi 경유 호출  
      await ApartmentStorage.saveFromApartmentApi(apt);  
      print('✅ 아파트 정보 저장 완료');  
    } catch (e) {  
      print('❌ /user/apartment 호출/파싱 에러: $e');  
    }  
  }  
}
